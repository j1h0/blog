[공유] [Java] Queue 예제

import java.util.LinkedList;

class Queue extends LinkedList { // Queue 클래스 선언
public Queue() { // 생성자
super();
}

public Object insert(Object o) { // 객체 추가 메소드(push)
addLast(o);
return(o);
}
public Object delete() { // 객체 삭제 메소드(pop)
return(removeFirst());
}
public Object front() { // Queue 의 front 위치를 반환하는 메소드
return(getFirst());
}
public Object rear() { // Queue 의 rear 위치를 반환하는 메소드
return(getLast());
}
public boolean empty() { // Queue 의 empty 여부
return(size() == 0); // true or false 를 리턴합니다.
}
}

public class Main {
public static void main(String[] args) {
Queue s = new Queue(); // Queue 객체생성(큐 생성)

// 객체를 insert함.
s.insert(new Integer(1)); // 객체를 push함.
System.out.println("pushed : 1");

s.insert(new Integer(3));
System.out.println("pushed : 3");

s.insert(new String("abc"));
System.out.println("pushed : abc");

System.out.println("\nqueue front: "+s.front()); // 큐의 정보 System.out.println("queue rear: "+s.rear()); // 큐의 정보
System.out.println();

// insert 완료

while(!s.empty()) { // 큐가 다 비워질때까지 반복.
System.out.println("pop: "+s.delete());
// delete하면서 화면에 출력.
}
}
}

출력결과는...

pushed : 1
pushed : 3
pushed : abc

queue front : 1
queue rear : abc

pop : 1
pop : 3
pop : abc
